Quick start

curl https://api.picbo.ai/v1/generate/photo \
  -H "Authorization: Bearer pk_live_xxxxxxxx" \
  -H "Content-Type: application/json" \
  -d '{
    "prompt": "cinematic product shot of a watch on marble",
    "style": "studio",
    "aspectRatio": "1:1"
  }'
const res = await fetch("https://api.picbo.ai/v1/generate/photo", {
  method: "POST",
  headers: {
    "Authorization": "Bearer pk_live_xxxxxxxx",
    "Content-Type": "application/json"
  },
  body: JSON.stringify({
    prompt: "cinematic product shot of a watch on marble",
    style: "studio",
    aspectRatio: "1:1"
  })
});
const job = await res.json();
import requests

res = requests.post(
    "https://api.picbo.ai/v1/generate/photo",
    headers={"Authorization": "Bearer pk_live_xxxxxxxx"},
    json={
        "prompt": "cinematic product shot of a watch on marble",
        "style": "studio",
        "aspectRatio": "1:1",
    },
)
job = res.json()
